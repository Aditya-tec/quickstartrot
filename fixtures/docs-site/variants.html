Getting Started

Install

npm i acme

Your first server

Let's write our first server:

// Require the framework and instantiate it

// ESM
import Acme from 'acme'

const acme = Acme({
  logger: true
})
// CommonJs
const acme = require('acme')({
  logger: true
})

// Declare a route
acme.get('/', function (request, reply) {
  reply.send({ hello: 'world' })
})

acme.listen({ port: 3000 })

Your first plugin

// ESM
import Acme from 'acme'
import routes from './routes.js'
const acme = Acme()
acme.register(routes)
// CommonJs
const acme = require('acme')()
acme.register(require('./routes'))
